const wait=(ms)=>new Promise(r=>setTimeout(r,ms));
const settle=async()=>{ for(let i=0;i<20;i++){ if(document.getAnimations().filter(a=>a.playState==='running' && a.effect?.target?.closest?.('main')).length===0) break; await wait(300);} };
const vw=document.documentElement.clientWidth;
const main=document.querySelector('[class*=_main_5xxpx]');
const over=()=>{const out=[...document.querySelectorAll('main *')].filter(e=>{const r=e.getBoundingClientRect(); if(!(r.width>0 && r.right>vw+1)) return false; for(let a=e.parentElement;a;a=a.parentElement){const cs=getComputedStyle(a); if(/auto|scroll|hidden|clip/.test(cs.overflowX) && a!==main && !a.matches('[class*=_app_]')) return false;} return true;}); return (main.scrollWidth>main.clientWidth?'SCROLL ':'')+out.slice(0,2).map(e=>(e.className||e.tagName).toString().slice(0,28)+' R'+Math.round(e.getBoundingClientRect().right)).join(', ');};
const res=[];
for (const t of [...document.querySelectorAll('nav[role=tablist][class*=stick] [role=tab]')].map(b=>b.textContent.trim())) { [...document.querySelectorAll('nav[role=tablist][class*=stick] [role=tab]')].find(b=>b.textContent.trim()===t).click(); await wait(400); await settle(); res.push(t+': '+(over()||'ok')); }
JSON.stringify({vw, res});
